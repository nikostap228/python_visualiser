🔍 Поиск в ширину (BFS)




Шаг: 0 / 0 |
Не посещены | В очереди | Посещены | Текущий 
from collections import deque

def bfs(graph, start):
    visited = [False] * len(graph)
    queue = deque([start])
    visited[start] = True  # Маркируем старт
    
    while queue:            # ← Пока очередь не пуста
        node = queue.popleft()  # Извлекаем узел
        # process(node)      # Обрабатываем
        
        for neighbor in graph[node]:
            if not visited[neighbor]:
                queue.append(neighbor)
                visited[neighbor] = True  # Маркируем при добавлении
    return visited
